# Run python with the dev example env

# 1.- Run nginx in front of Flask for protected files
$env:FILES_LOCAL_ROOT ="C:/CADEXPORT"
echo "${env:FILES_LOCAL_ROOT}"
docker run --rm -d --name tinymrp-nginx-dev -p 5001:80 -v "${PWD}/docker/nginx/nginx.dev.conf:/etc/nginx/nginx.conf:ro" -v "${env:FILES_LOCAL_ROOT}:/data/deliverables:ro" nginx:1.27-alpine

# 2. - Launch mongo
# 3.- Launch the app using the env.dev.example check there theaddress for mongo
